O83 — POP/PUSH pair elimination
================================

== C test case ==================================================================

// O83 — POP/PUSH pair elimination test case.
//
// Uses a simplified Sieve of Eratosthenes kernel (SIZE=200) which is known
// to generate redundant POP/PUSH pairs in the outer loop maintenance code.

typedef unsigned short u16;
typedef unsigned char  u8;

static u8 flags[200];

u16 sieve_count(u16 n) {
    u16 i, k, count, i_sq;
    for (k = 0; k < n; k++) flags[k] = 0;
    count = (u16)(n - 2);
    i_sq  = 4;
    for (i = 2; i_sq < n; ++i) {
        if (!flags[i]) {
            for (k = i_sq; k < n; k = (u16)(k + i)) {
                if (!flags[k]) count = (u16)(count - 1);
                flags[k] = 1;
            }
        }
        i_sq = (u16)(i_sq + i + i + 1);
    }
    return count;
}

int main(void) {
    u16 c = sieve_count(200);
    return (int)((u8)c ^ (u8)(c >> 8));
}

== c8080 — relevant outer-loop latch (i8080 syntax) ============================
; (c8080 emits all variables in static memory; outer-loop update shown)
; Source lines 29-30: i_sq = i_sq + i + i + 1; ++i

; c8080 uses LHLD/SHLD for every 16-bit variable read/write.
; i_sq += i
        LHLD    sieve_count_i_sq        ; 20cc  3B  load i_sq
        XCHG                            ;  4cc  1B  i_sq -> DE
        LHLD    sieve_count_i           ; 20cc  3B  load i
        DAD     D                       ; 12cc  1B  HL = i + i_sq
        XCHG                            ;  4cc  1B  partial sum -> DE
; i_sq += i again  (i_sq + 2i)
        LHLD    sieve_count_i           ; 20cc  3B  load i
        DAD     D                       ; 12cc  1B  HL = i_sq + 2i
; i_sq += 1
        INX     H                       ;  8cc  1B  HL = i_sq + 2i + 1
        SHLD    sieve_count_i_sq        ; 20cc  3B  store i_sq
; i++
        LHLD    sieve_count_i           ; 20cc  3B  load i
        INX     H                       ;  8cc  1B  HL = i + 1
        SHLD    sieve_count_i           ; 20cc  3B  store i
        JMP     l_3                     ; 12cc  3B  back to outer header

; c8080 outer-loop latch cycles (per iteration): 180cc
; c8080 outer-loop latch size:  28 bytes

== v6clang_old — outer-loop latch (.LBB16_8, BEFORE O83) =======================

; Two redundant POP/PUSH pairs are present (marked <<<).

.LBB16_8:                       ; outer-loop latch header

.LLo61_10:
        LXI     B, 0             ; 12cc  3B  O61 patched: BC = i_sq partial
